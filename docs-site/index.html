import { fetchSSE } from '@stackline/sse';

for await (const event of fetchSSE('/api/generate', {
  method: 'POST',
  body: JSON.stringify({ prompt: 'Hello', stream: true }),
  idleTimeout: 45_000,
  retry: { retries: 3 }
})) {
  if (event.data === '[DONE]') break;
  render(JSON.parse(event.data));
}
